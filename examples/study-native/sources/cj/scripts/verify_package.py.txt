#!/usr/bin/env python3
import csv,hashlib,json,yaml
from pathlib import Path
root=Path(__file__).resolve().parents[1]
expected=json.loads((root/'manifest_sha256.json').read_text(encoding='utf-8'))
actual={str(p.relative_to(root)):hashlib.sha256(p.read_bytes()).hexdigest() for p in sorted(root.rglob('*')) if p.is_file() and p.name!='manifest_sha256.json' and 'actual_engine_previews' not in p.relative_to(root).parts and '__pycache__' not in p.relative_to(root).parts}
assert expected==actual, 'File hash/manifest mismatch'
configs=list((root/'configs').glob('*.yaml'))
assert len(configs)==(24 if (root/'rubric_12_traits.yaml').exists() else 6)
for file in configs:
 cfg=yaml.safe_load(file.read_text());assert cfg['schema_version']==3 and cfg['prompt']['structured_output']=='text'
 for path in ['input','rubric'] if cfg['task_kind']=='rubric' else ['input','criteria','pairs']:
  assert (file.parent/cfg[path]).exists(),(file,path)
 assert (file.parent/cfg['prompt']['file']).exists()
if (root/'rubric_12_traits.yaml').exists():
 r=yaml.safe_load((root/'rubric_12_traits.yaml').read_text());assert len(r['traits'])==12
 for t in r['traits']:
  assert [v['label'] for v in t['levels']]==['0','1','2'];assert t['levels'][0]['description']=='Criterion not yet fulfilled.'
else:
 c=yaml.safe_load((root/'criteria_13_traits.yaml').read_text())['criteria'];assert len(c)==13 and c[-1]['trait_id']=='overall_writing_quality'
 for file in (root/'pairs_demo.jsonl',root/'pairs_reversal_demo.jsonl'):
  pairs=[json.loads(x) for x in file.read_text().splitlines()];assert len(pairs)==13 and set(p['trait_id'] for p in pairs)==set(x['trait_id'] for x in c)
 for file in (root/'rendered_illustrations').glob('*.txt'):
  text=file.read_text();assert 'student_opaque_' not in text and 'essay_01:' in text and 'essay_02:' in text
print('PASS: manifest integrity, config paths, criterion/rubric membership and illustrative response-ID checks')
